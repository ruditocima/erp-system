Panduan Setup Supabase Database & GitHub Pages

Ikuti langkah di bawah ini untuk menghubungkan WMS ini ke database cloud gratis dan hosting online.

1. Skema Tabel SQL di Supabase

Jalankan perintah SQL berikut di SQL Editor Supabase Anda:

-- Tabel Master Barang
CREATE TABLE master_barang (
    id SERIAL PRIMARY KEY,
    kode VARCHAR(50) UNIQUE NOT NULL,
    nama VARCHAR(150) NOT NULL,
    kategori VARCHAR(100),
    jenis VARCHAR(100),
    satuan VARCHAR(20)
);

-- Tabel Master Gudang
CREATE TABLE master_gudang (
    id SERIAL PRIMARY KEY,
    kode VARCHAR(50) UNIQUE NOT NULL,
    nama VARCHAR(150) NOT NULL,
    tipe VARCHAR(50),
    lokasi TEXT,
    pic VARCHAR(100)
);

-- Tabel Project
CREATE TABLE master_project (
    id SERIAL PRIMARY KEY,
    kode_project VARCHAR(50) UNIQUE NOT NULL,
    periode VARCHAR(50),
    type VARCHAR(100),
    region VARCHAR(100),
    no_pr_po VARCHAR(100),
    po_plan VARCHAR(50),
    po_final VARCHAR(50),
    status_po VARCHAR(50),
    permit VARCHAR(50),
    status_snd VARCHAR(50),
    status_project VARCHAR(50),
    status_doc VARCHAR(50)
);

-- Tabel Stok Gudang (Update: Termasuk metrik transfer & mutasi)
CREATE TABLE stok_gudang (
    id SERIAL PRIMARY KEY,
    gudang VARCHAR(150),
    kode_barang VARCHAR(50),
    nama_barang VARCHAR(150),
    qty INT DEFAULT 0,
    masuk INT DEFAULT 0,
    keluar INT DEFAULT 0,
    t_masuk INT DEFAULT 0,
    t_keluar INT DEFAULT 0
);

-- Nonaktifkan RLS agar terhindar dari Error 401 Unauthorized
ALTER TABLE master_barang DISABLE ROW LEVEL SECURITY;
ALTER TABLE master_gudang DISABLE ROW LEVEL SECURITY;
ALTER TABLE master_project DISABLE ROW LEVEL SECURITY;
ALTER TABLE stok_gudang DISABLE ROW LEVEL SECURITY;
